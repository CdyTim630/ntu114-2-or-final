\section{Motivation}
\begin{frame}{Background \& Motivation}
\begin{itemize}
  \item Convenience stores are the most frequent real-time meal channel for students and office workers.
  \item One pick juggles many goals at once: \textbf{cost, calories, protein, sugar, sodium, satiety}.
  \item Bundle promotions (e.g.\ rice ball + drink) change the \textbf{real} cost --- the cheapest combo is not obvious.
  \item Ranking by a single metric is not enough $\Rightarrow$ we need \textbf{constrained optimization}.
\end{itemize}
\end{frame}

\begin{frame}{Decision Maker: Inputs, Outputs, Fallback}
\textbf{Decision maker:} the user. \medskip

\textbf{Input}
\begin{itemize}
  \item Scenario: light / regular / late-night $\;\cdot\;$ Goal: maintain / cut / bulk
  \item Profile: sex, age, height, weight, activity $\;\cdot\;$ Budget cap $B$
\end{itemize}
\textbf{Output}
\begin{itemize}
  \item Recommended single items + promo bundles, with expected nutrition and total cost
\end{itemize}
\textbf{Fallback}
\begin{itemize}
  \item If infeasible: the closest alternative via soft-constraint relaxation
\end{itemize}
\end{frame}

\section{Model}
\begin{frame}{Preprocessing: Turning User Profile into Meal Limits}
We convert each user profile into personalized calorie, protein, and item-count limits for a single meal.
\begin{center}
\setlength{\fboxsep}{6pt}
\colorbox{green!10}{\color{green!45!black}\textbf{Profile $\to$ BMR $\to$ TDEE $\to$ goal-adjusted calories $\to$ single-meal limits}}
\end{center}
\[
\text{BMR} = \begin{cases}10W+6.25H-5A+5, & \text{male}\\10W+6.25H-5A-161, & \text{female}\end{cases}
\;\Rightarrow\; \text{TDEE} = \text{BMR}\cdot AF
\]
\[
\text{Target}_{\text{Cal}} = \text{TDEE} + \delta_{\text{goal}},\quad
\delta\in\{0,-500,+300\}\;(\text{maintain/cut/bulk})
\]
\begin{center}
\textbf{Scenario-based meal scaling}\\[3pt]
\begin{tabular}{lcccc}
\toprule
Scenario & $\rho^{\min}$ & $\rho^{\max}$ & $\rho$ (protein) & $N^{\max}$ \\
\midrule
Light & 0.15 & 0.25 & 0.20 & 3 \\
Regular & 0.30 & 0.45 & 0.40 & 5 \\
Late-night & 0.05 & 0.15 & 0.10 & 3 \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\begin{frame}{MILP --- Core Decision Variables}
\textbf{Binary variables}
\[
a_i \in \{0,1\}\;(\text{single item } i),\qquad
y_k \in \{0,1\}\;(\text{bundle } k)
\]
\begin{itemize}
  \item Auxiliary variable $x_i$ prevents the same item from being selected twice:
  \(\displaystyle x_i = a_i + \sum_{k:\,i\in S_k} y_k \;\le\; 1\)
  \item Non-negative slacks $s_P, s^{\pm}_{F}, s^{\pm}_{C}, s_{\text{Sug}}, s_{\text{Sod}} \ge 0$ relax the soft targets
\end{itemize}
\end{frame}

\begin{frame}{Tiered Memory: Penalizing Repetition (MLFQ-style)}
\textbf{Inspired by MLFQ, recently recommended items receive stronger penalties, and older items gradually fade.}
\medskip

\begin{itemize}
  \item Tier weights $\boldsymbol\omega=(5,2.5,1,0.3)$ decay with recency; a re-picked item jumps back to the top tier
  \item $h_i$ = \textbf{same-item} penalty;\; $g_i$ = \textbf{same-category} penalty (e.g.\ all soy milks)
\end{itemize}
\[ h_i = \sum_r \omega_{t(r)}\,\mathbf{1}(i \in \text{rec}_r),\qquad
   g_i = \sum_r \omega_{t(r)}\,\mathbf{1}(\mathrm{role}(\text{rec}_r)=\mathrm{role}(i)) \]
\end{frame}

\begin{frame}{MILP --- Objective \& Constraints}
\textbf{Objective (minimize):}
\[
\alpha \underbrace{\left(\sum c_i a_i + \sum c_k^{\text{promo}} y_k\right)}_{\text{total cost}}
- \beta \sum pro_i x_i
+ \gamma \sum h_i x_i + \gamma_{\text{cat}} \sum g_i x_i
+ \sum_{*} \lambda_* s_*
\]
\begin{itemize}
  \item \textbf{Hard constraints:} budget, calorie range, mutual exclusion, and item count
  \item \textbf{Meal-structure constraints:} drink/dessert caps and regular-meal composition
  \item \textbf{Safety constraint:} sodium $\le 1.5\times$ recommended cap
  \item \textbf{Soft constraints (slack):} protein floor, fat/carb ranges, sugar/sodium caps
\end{itemize}
\smallskip
{\small\emph{Weights control the relative importance of cost, protein, repetition, and soft-constraint violations.}}
\end{frame}

\section{Algorithm: GRASP-LS}
\begin{frame}{Why a Heuristic in Addition to Gurobi?}
\begin{itemize}
  \item The project requires our \textbf{own algorithm}, benchmarked against the exact optimum
  \item \textbf{License-free} and edge-deployable; extensible to multiple store chains
  \item \textbf{Anytime}: can stop early and still return the best meal found --- suitable for interactive deployment with controllable time limits
\end{itemize}
\end{frame}

\begin{frame}{GRASP-LS --- Algorithm Flowchart}
\centering
\begin{tikzpicture}[
  font=\scriptsize, >=Stealth, node distance=4mm,
  box/.style={rectangle, rounded corners, draw=green!45!black, thick,
              fill=green!8, align=center, inner sep=3pt, text width=72mm, minimum height=7mm},
  term/.style={rectangle, rounded corners=9pt, draw=green!45!black, thick,
               fill=green!22, align=center, inner sep=3pt, text width=58mm, minimum height=6mm},
  dec/.style={diamond, aspect=2.4, draw=orange!75!black, thick, fill=orange!12,
              align=center, inner sep=0pt}]
  \node[term] (in) {Input: items, meal target, budget $B$};
  \node[box, below=of in] (grc) {\textbf{GRC} --- build RCL by score $\sigma$\\(main first; drink/dessert caps; combos as super-items)};
  \node[box, below=of grc] (ls) {\textbf{Local Search} (first-improvement)\\swap / drop / add / combo-swap};
  \node[box, below=of ls] (up) {Update best solution \& pool};
  \node[dec, below=of up] (dec) {more\\restarts?};
  \node[term, below=of dec] (out) {Return best meal / sample $\epsilon$-best pool};
  \draw[->] (in) -- (grc);
  \draw[->] (grc) -- (ls);
  \draw[->] (ls) -- (up);
  \draw[->] (up) -- (dec);
  \draw[->] (dec) -- node[right]{no} (out);
  \draw[->] (dec.east) -- ++(3.3cm,0) |- (grc.east);
  \node[align=left] at ($(dec.east)+(3.65cm,1.7cm)$) {yes\\($\times25$)};
\end{tikzpicture}

\vspace{1mm}
{\scriptsize Deployment samples the $\epsilon$-best pool ($\epsilon=8$) for variety; falls back to MILP if a tight budget leaves GRASP-LS infeasible.}
\end{frame}

\begin{frame}{Composite Score: How the Heuristic Ranks Candidates}
\[
\sigma(p) = \beta\cdot pro_p - 0.06\alpha\cdot c_p - \gamma h_p
\;-\;0.6\tfrac{cal_p}{\text{Cal}^{\max}_m}
\;-\;\lambda_{\text{Sod}}\tfrac{sod_p}{\text{Sodium}^{\max}_m}
\;-\;\lambda_{\text{Sug}}\tfrac{sug_p}{\text{Sugar}^{\max}_m}
+ B_{\text{cat}}(p)
\]
\begin{itemize}
  \item Rewards \textbf{protein} and \textbf{structural fit} (mains, protein sources)
  \item Penalizes \textbf{cost, excess calories, sodium, sugar, and repetition}
  \item Treats \textbf{bundles as super-items}, prioritized by discount
\end{itemize}
\end{frame}

\section{Conclusion}
\begin{frame}{Conclusions --- Academic Contributions}
\begin{itemize}
  \item \textbf{MILP formulation} with promo bundles, personalized nutrition, meal structure, and a diversity penalty
  \item \textbf{GRASP-LS}: high feasibility with small optimality gaps (\textbf{7.1\%} / \textbf{4.9\%}), anytime
  \item \textbf{Real-data validation}: 488 items + a working web prototype
\end{itemize}
\end{frame}

\begin{frame}{Conclusions --- Practical Insights}
\begin{itemize}
  \item Simple greedy ranking \textbf{fails} under realistic constraints (\alert{0\% feasible})
  \item \textbf{Role tagging} makes recommendations meal-like, not nutrient piles
  \item \textbf{Real value}: compared with a simulated blind pick, all-target attainment increases from $\sim$4\% to 100\% at comparable cost
  \item \textbf{Personalization is practical}: $\beta$ controls the cost--protein preference, while diversity can be added at low cost
\end{itemize}
\end{frame}
